# Phase 6 — Lint Tier 1+2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a deterministic lint system (no LLM) that detects 8 types of wiki health issues across two tiers (provenance/integrity + navigability), exposes them through API + UI panel with per-issue fix actions, and persists reports as JSON.

**Architecture:** Backend gains `services/lint/` module (checks → runner → fix_applier), `models/lint_models.py`, and a `/api/lint/*` router. WikiStore gets minimal delete/update helpers needed by fixers. Frontend adds `LintPanel.tsx` with a Run Now button, filterable issue table, and per-issue Apply/Reject/Won't-fix actions. Lint reports are stored as `wiki_data/wiki/lint_reports/<TS>.json`.

**Tech Stack:** Python 3.9, FastAPI, Pydantic v2, pytest. Frontend: React 19, TypeScript, Tailwind CSS, lucide-react.

**Spec:** `docs/superpowers/specs/2026-05-07-phase-6-lint-design.md`

```python
from datetime import datetime
from typing import Any, Dict, List, Literal, Optional

from pydantic import BaseModel, Field


IssueSeverity = Literal["error", "warning", "info"]

IssueType = Literal[
    # Tier 1 (error)
    "provenance_missing",
    "dangling_source_ref",
    "dangling_node_ref",
    "corrupted_jsonl",
    # Tier 2 (warning)
    "orphan_entity",
    "orphan_concept",
    "empty_questions",
    "empty_description",
]

IssueStatus = Literal["open", "fixed", "rejected", "wontfix"]


class LintIssue(BaseModel):
    id: str
    type: IssueType
    severity: IssueSeverity
    location: str
    description: str
    suggested_fix: Optional[str] = None
    auto_fixable: bool = False
    payload: Dict[str, Any] = Field(default_factory=dict)
    status: IssueStatus = "open"


class LintReport(BaseModel):
    id: str
    created_at: datetime
    issues: List[LintIssue]
    counts_by_severity: Dict[str, int]
    counts_by_type: Dict[str, int]
```

```python
import json
from typing import List, Set

from models.lint_models import LintIssue
from services.wiki_store import WikiStore


REQUIRED_CLAIM_FIELDS = {"id", "text", "subject_id", "source_id", "span", "confidence", "polarity"}
REQUIRED_EDGE_FIELDS = {"id", "from", "to", "relation", "source_id", "span", "confidence"}


def _read_jsonl_dicts(path) -> List[tuple]:
    """Yield (line_no, parsed_dict_or_None_on_error, raw_line) per non-blank line."""
    if not path.exists():
        return []
    out = []
    for i, raw in enumerate(path.read_text(encoding="utf-8").splitlines(), start=1):
        if not raw.strip():
            continue
        try:
            out.append((i, json.loads(raw), raw))
        except json.JSONDecodeError:
            out.append((i, None, raw))
    return out


def check_provenance_missing(store: WikiStore) -> List[LintIssue]:
    issues: List[LintIssue] = []

    claims_path = store.root / "wiki" / "claims.jsonl"
    for line_no, parsed, _raw in _read_jsonl_dicts(claims_path):
        if parsed is None:
            continue  # corrupted lines handled in check_corrupted_jsonl
        missing = REQUIRED_CLAIM_FIELDS - set(parsed.keys())
        cid = parsed.get("id", f"line:{line_no}")
        if missing:
            issues.append(LintIssue(
                id=f"provenance_missing:claim:{cid}",
                type="provenance_missing",
                severity="error",
                location=f"claims.jsonl:{cid}",
                description=f"Claim {cid} is missing required fields: {sorted(missing)}",
                suggested_fix=None,
                auto_fixable=False,
                payload={"kind": "claim", "claim_id": cid, "missing": sorted(missing)},
            ))

    edges_path = store.root / "wiki" / "edges.jsonl"
    for line_no, parsed, _raw in _read_jsonl_dicts(edges_path):
        if parsed is None:
            continue
        missing = REQUIRED_EDGE_FIELDS - set(parsed.keys())
        eid = parsed.get("id", f"line:{line_no}")
        if missing:
            issues.append(LintIssue(
                id=f"provenance_missing:edge:{eid}",
                type="provenance_missing",
                severity="error",
                location=f"edges.jsonl:{eid}",
                description=f"Edge {eid} is missing required fields: {sorted(missing)}",
                suggested_fix=None,
                auto_fixable=False,
                payload={"kind": "edge", "edge_id": eid, "missing": sorted(missing)},
            ))

    return issues


def _collect_source_ids(store: WikiStore) -> Set[str]:
    ids: Set[str] = set()
    for slug in store.list_sources():
        s = store.read_source(slug)
        if s:
            ids.add(s.id)
    return ids


def _collect_node_ids(store: WikiStore) -> Set[str]:
    """Returns prefixed node IDs ('entity:e_001', 'concept:c_001')."""
    from services.wiki_store import ENTITIES_DIR, CONCEPTS_DIR
    ids: Set[str] = set()

    entities_dir = store.root / "wiki" / ENTITIES_DIR
    if entities_dir.exists():
        for type_dir in entities_dir.iterdir():
            if not type_dir.is_dir():
                continue
            for path in type_dir.glob("*.md"):
                ent = store.read_entity(path.stem, type_dir.name)
                if ent:
                    ids.add(f"entity:{ent.id}")

    concepts_dir = store.root / "wiki" / CONCEPTS_DIR
    if concepts_dir.exists():
        for path in concepts_dir.glob("*.md"):
            c = store.read_concept(path.stem)
            if c:
                ids.add(f"concept:{c.id}")

    return ids


def check_dangling_source_ref(store: WikiStore) -> List[LintIssue]:
    valid_sources = _collect_source_ids(store)
    issues: List[LintIssue] = []

    for c in store.read_all_claims():
        if c.source_id not in valid_sources:
            issues.append(LintIssue(
                id=f"dangling_source_ref:claim:{c.id}",
                type="dangling_source_ref",
                severity="error",
                location=f"claims.jsonl:{c.id}",
                description=f"Claim {c.id} references missing source {c.source_id}",
                suggested_fix="Delete claim",
                auto_fixable=True,
                payload={"kind": "claim", "claim_id": c.id, "source_id": c.source_id},
            ))

    for e in store.read_all_edges():
        if e.source_id not in valid_sources:
            issues.append(LintIssue(
                id=f"dangling_source_ref:edge:{e.id}",
                type="dangling_source_ref",
                severity="error",
                location=f"edges.jsonl:{e.id}",
                description=f"Edge {e.id} references missing source {e.source_id}",
                suggested_fix="Delete edge",
                auto_fixable=True,
                payload={"kind": "edge", "edge_id": e.id, "source_id": e.source_id},
            ))

    return issues


def check_dangling_node_ref(store: WikiStore) -> List[LintIssue]:
    valid_nodes = _collect_node_ids(store)
    issues: List[LintIssue] = []

    for e in store.read_all_edges():
        for endpoint, value in (("from", e.from_), ("to", e.to)):
            if value not in valid_nodes:
                issues.append(LintIssue(
                    id=f"dangling_node_ref:edge:{e.id}:{endpoint}",
                    type="dangling_node_ref",
                    severity="error",
                    location=f"edges.jsonl:{e.id}",
                    description=f"Edge {e.id} {endpoint}={value} points to missing node",
                    suggested_fix="Delete edge",
                    auto_fixable=True,
                    payload={"edge_id": e.id, "endpoint": endpoint, "missing_id": value},
                ))

    return issues


def check_corrupted_jsonl(store: WikiStore) -> List[LintIssue]:
    issues: List[LintIssue] = []
    for filename in ("claims.jsonl", "edges.jsonl"):
        path = store.root / "wiki" / filename
        if not path.exists():
            continue
        for line_no, raw in enumerate(path.read_text(encoding="utf-8").splitlines(), start=1):
            if not raw.strip():
                continue
            try:
                json.loads(raw)
            except json.JSONDecodeError as exc:
                issues.append(LintIssue(
                    id=f"corrupted_jsonl:{filename}:{line_no}",
                    type="corrupted_jsonl",
                    severity="error",
                    location=f"{filename}:{line_no}",
                    description=f"Line {line_no} of {filename} is not valid JSON: {exc.msg}",
                    suggested_fix="Delete corrupt line (with .bak backup)",
                    auto_fixable=True,
                    payload={"file": filename, "line_no": line_no},
                ))
    return issues
```

```python
from typing import List, Set

from models.lint_models import LintIssue
from services.wiki_store import CONCEPTS_DIR, ENTITIES_DIR, WikiStore


MIN_DESCRIPTION_CHARS = 30


def _all_entities(store: WikiStore):
    base = store.root / "wiki" / ENTITIES_DIR
    if not base.exists():
        return
    for type_dir in base.iterdir():
        if not type_dir.is_dir():
            continue
        for path in type_dir.glob("*.md"):
            ent = store.read_entity(path.stem, type_dir.name)
            if ent:
                yield path.stem, ent


def _all_concepts(store: WikiStore):
    base = store.root / "wiki" / CONCEPTS_DIR
    if not base.exists():
        return
    for path in base.glob("*.md"):
        c = store.read_concept(path.stem)
        if c:
            yield path.stem, c


def _all_sources(store: WikiStore):
    for slug in store.list_sources():
        s = store.read_source(slug)
        if s:
            yield slug, s


def check_orphan_entity(store: WikiStore) -> List[LintIssue]:
    edges = store.read_all_edges()
    referenced: Set[str] = set()
    for e in edges:
        referenced.add(e.from_)
        referenced.add(e.to)

    claims_subjects = {c.subject_id for c in store.read_all_claims()}

    issues: List[LintIssue] = []
    for slug, ent in _all_entities(store):
        node_id = f"entity:{ent.id}"
        if node_id in referenced or node_id in claims_subjects:
            continue
        issues.append(LintIssue(
            id=f"orphan_entity:{ent.id}",
            type="orphan_entity",
            severity="warning",
            location=node_id,
            description=f"Entity {ent.id} ({ent.type}/{slug}) has no edges and no claims",
            suggested_fix="Delete entity",
            auto_fixable=True,
            payload={"entity_id": ent.id, "slug": slug, "entity_type": ent.type},
        ))
    return issues


def check_orphan_concept(store: WikiStore) -> List[LintIssue]:
    incoming: Set[str] = {e.to for e in store.read_all_edges()}
    issues: List[LintIssue] = []
    for slug, c in _all_concepts(store):
        node_id = f"concept:{c.id}"
        if node_id in incoming:
            continue
        issues.append(LintIssue(
            id=f"orphan_concept:{c.id}",
            type="orphan_concept",
            severity="warning",
            location=node_id,

                severity="warning",
                location=node_id,
                description=f"Concept {c.id} has empty questions[]",
                suggested_fix=None,
                auto_fixable=False,

            node_id = f"source:{s.id}"
            issues.append(LintIssue(

                auto_fixable=False,

def check_empty_description(store: WikiStore) -> List[LintIssue]:
    issues: List[LintIssue] = []

        if len(ent.description.strip()) < MIN_DESCRIPTION_CHARS:
            node_id = f"entity:{ent.id}"
            issues.append(LintIssue(
                id=f"empty_description:entity:{ent.id}",
                type="empty_description",
                severity="warning",
                location=node_id,

                suggested_fix="Edit description inline",
                auto_fixable=True,

        if len(c.definition.strip()) < MIN_DESCRIPTION_CHARS:
            node_id = f"concept:{c.id}"
            issues.append(LintIssue(
                id=f"empty_description:concept:{c.id}",
                type="empty_description",
                severity="warning",
                location=node_id,
                description=f"Concept {c.id} definition is shorter than {MIN_DESCRIPTION_CHARS} chars",
                suggested_fix="Edit definition inline",
                auto_fixable=True,
                payload={"node_id": node_id, "kind": "concept", "slug": slug},
            ))

    for slug, s in _all_sources(store):
        if len(s.summary.strip()) < MIN_DESCRIPTION_CHARS:
            node_id = f"source:{s.id}"
            issues.append(LintIssue(
                id=f"empty_description:source:{s.id}",
                type="empty_description",
                severity="warning",
                location=node_id,
                description=f"Source {s.id} summary is shorter than {MIN_DESCRIPTION_CHARS} chars",

                auto_fixable=False,
                payload={"node_id": node_id, "kind": "source", "slug": slug},
            ))
```

```python
    # ── deletes / updates (Phase 6 lint fixers) ───────────────────────────

    def _rewrite_jsonl_skipping(self, path: Path, predicate) -> int:
        """Rewrite path keeping only lines where predicate(parsed_dict) is True. Returns count removed."""
        if not path.exists():
            return 0
        kept: List[str] = []
        removed = 0
        for raw in path.read_text(encoding="utf-8").splitlines():
            if not raw.strip():
                continue
            try:
                parsed = json.loads(raw)
            except json.JSONDecodeError:
                kept.append(raw)
                continue
            if predicate(parsed):
                kept.append(raw)
            else:
                removed += 1
        new_content = "\n".join(kept) + ("\n" if kept else "")
        self._atomic_write(path, new_content)
        return removed

    def delete_claim(self, claim_id: str) -> bool:
        path = self._claims_path()
        removed = self._rewrite_jsonl_skipping(path, lambda d: d.get("id") != claim_id)
        return removed > 0

    def delete_edge(self, edge_id: str) -> bool:
        path = self._edges_path()
        removed = self._rewrite_jsonl_skipping(path, lambda d: d.get("id") != edge_id)
        return removed > 0

    def delete_jsonl_line(self, path: Path, line_no: int) -> None:
        """Remove the 1-indexed line, preserve a `.bak` copy of the original file."""
        original = path.read_text(encoding="utf-8")
        backup = path.with_suffix(path.suffix + ".bak")
        backup.write_text(original, encoding="utf-8")
        lines = original.splitlines()
        if 1 <= line_no <= len(lines):
            del lines[line_no - 1]
        new_content = "\n".join(lines) + ("\n" if lines else "")
        self._atomic_write(path, new_content)

    def delete_entity(self, slug: str, entity_type: str) -> bool:
        path = self._entity_path(slug, entity_type)
        if not path.exists():
            return False
        path.unlink()
        return True

    def delete_concept(self, slug: str) -> bool:
        path = self._concept_path(slug)
        if not path.exists():
            return False
        path.unlink()
        return True

    def update_entity_description(self, slug: str, entity_type: str, new_description: str) -> None:
        ent = self.read_entity(slug, entity_type)
        if ent is None:
            raise FileNotFoundError(f"Entity not found: {entity_type}/{slug}")
        updated = ent.model_copy(update={"description": new_description})
        self.write_entity(updated, slug)

    def update_concept_description(self, slug: str, new_definition: str) -> None:
        c = self.read_concept(slug)
        if c is None:
            raise FileNotFoundError(f"Concept not found: {slug}")
        updated = c.model_copy(update={"definition": new_definition})
        self.write_concept(updated, slug)
```

> Note: `_rewrite_jsonl_skipping` writes content using `_atomic_write`, which uses `path.with_suffix(path.suffix + ".tmp")` then `.replace(path)` — safe across Windows and POSIX.

```python
    check_dangling_source_ref,
    check_provenance_missing,
)
from services.wiki_log import append_log_entry
from services.wiki_store import WikiStore


def _reports_dir(store: WikiStore) -> Path:
    d = store.root / "wiki" / "lint_reports"
    d.mkdir(parents=True, exist_ok=True)
    return d


def _make_report_id(now: datetime) -> str:
    return now.strftime("%Y-%m-%dT%H-%M-%S")


def _aggregate_issues(issues: List[LintIssue]) -> tuple[Dict[str, int], Dict[str, int]]:
    sev = {"error": 0, "warning": 0, "info": 0}
    by_type: Dict[str, int] = {}
    for i in issues:
        sev[i.severity] = sev.get(i.severity, 0) + 1
        by_type[i.type] = by_type.get(i.type, 0) + 1
    return sev, by_type


def run_lint(store: WikiStore) -> LintReport:
    issues: List[LintIssue] = []
    issues.extend(check_corrupted_jsonl(store))
    issues.extend(check_provenance_missing(store))
    issues.extend(check_dangling_source_ref(store))
    issues.extend(check_dangling_node_ref(store))
    issues.extend(check_orphan_entity(store))
    issues.extend(check_orphan_concept(store))
    issues.extend(check_empty_questions(store))
    issues.extend(check_empty_description(store))

    now = datetime.now()
    report_id = _make_report_id(now)
    sev, by_type = _aggregate_issues(issues)

    report = LintReport(
        id=report_id,
        created_at=now,
        issues=issues,
        counts_by_severity=sev,
        counts_by_type=by_type,
    )

    save_report(store, report)
    append_log_entry(
        store,
        op="lint",
        source_id=None,
        details=f"run {report_id} | found: {len(issues)} ({sev['error']} errors, {sev['warning']} warnings)",
        timestamp=now,
    )
    return report


def save_report(store: WikiStore, report: LintReport) -> None:
    path = _reports_dir(store) / f"{report.id}.json"
    payload = report.model_dump(mode="json")
    store._atomic_write(path, json.dumps(payload, indent=2, ensure_ascii=False))


def load_report(store: WikiStore, report_id: str) -> LintReport:
    path = _reports_dir(store) / f"{report_id}.json"
    return LintReport.model_validate_json(path.read_text(encoding="utf-8"))


def list_reports(store: WikiStore) -> List[Dict[str, Any]]:
    out: List[Dict[str, Any]] = []
    for p in sorted(_reports_dir(store).glob("*.json"), reverse=True):
        try:
            data = json.loads(p.read_text(encoding="utf-8"))
        except json.JSONDecodeError:
            continue
        out.append({
            "id": data.get("id", p.stem),
            "created_at": data.get("created_at"),
            "counts_by_severity": data.get("counts_by_severity", {}),
            "total": len(data.get("issues", [])),
        })
    return out


def update_issue_status(
    store: WikiStore,
    report_id: str,
    issue_id: str,
    new_status: IssueStatus,
) -> LintReport:
    report = load_report(store, report_id)
    found = False
    for i in report.issues:
        if i.id == issue_id:
            i.status = new_status
            found = True
            break
    if not found:
        raise KeyError(f"Issue {issue_id} not found in report {report_id}")
    save_report(store, report)
    return report
```

```python
from typing import Any, Dict, Optional

from models.lint_models import LintIssue
from services.wiki_store import WikiStore


class FixError(Exception):
    """Raised when a fix cannot be applied (not auto-fixable, missing payload, etc.)."""


def apply_fix(
    store: WikiStore,
    issue: LintIssue,
    edit_payload: Optional[Dict[str, Any]] = None,
) -> None:
    if not issue.auto_fixable:
        raise FixError(f"Issue {issue.id} ({issue.type}) is not auto-fixable")

    t = issue.type
    p = issue.payload

    if t == "dangling_source_ref":
        if p.get("kind") == "claim":
            store.delete_claim(p["claim_id"])
        elif p.get("kind") == "edge":
            store.delete_edge(p["edge_id"])
        else:
            raise FixError(f"Unknown payload.kind for dangling_source_ref: {p.get('kind')}")
        return

    if t == "dangling_node_ref":
        store.delete_edge(p["edge_id"])
        return

    if t == "corrupted_jsonl":
        path = store.root / "wiki" / p["file"]
        store.delete_jsonl_line(path, p["line_no"])
        return

    if t == "orphan_entity":
        store.delete_entity(p["slug"], p["entity_type"])
        return

    if t == "orphan_concept":
        store.delete_concept(p["slug"])
        return

    if t == "empty_description":
        if not edit_payload or "description" not in edit_payload:
            raise FixError("empty_description fix requires edit_payload['description']")
        new_text = edit_payload["description"]
        if p.get("kind") == "entity":
            store.update_entity_description(p["slug"], p["entity_type"], new_text)
        elif p.get("kind") == "concept":
            store.update_concept_description(p["slug"], new_text)
        else:
            raise FixError(f"empty_description fix not implemented for kind={p.get('kind')}")
        return

    raise FixError(f"No deterministic fix for type {t} (Phase 6)")
```

```python
import os
from pathlib import Path
from typing import Any, Dict, List, Literal, Optional

from fastapi import APIRouter, HTTPException
from pydantic import BaseModel

from models.lint_models import LintReport
from services.lint.fix_applier import FixError, apply_fix
from services.lint.runner import (
    list_reports,
    load_report,
    run_lint,
    update_issue_status,
)
from services.wiki_store import WikiStore


router = APIRouter(prefix="/api/lint", tags=["lint"])

_WIKI_DATA_DIR = Path(os.getenv("WIKI_DATA_DIR", "wiki_data/default"))


def _get_store() -> WikiStore:
    return WikiStore(root=_WIKI_DATA_DIR)


class RunLintBody(BaseModel):
    pass


class FixBody(BaseModel):
    report_id: str
    issue_id: str
    action: Literal["apply", "reject", "wontfix"]
    edit_payload: Optional[Dict[str, Any]] = None


@router.post("/run", response_model=LintReport)
def post_run_lint(_body: RunLintBody = RunLintBody()):
    store = _get_store()
    return run_lint(store)


@router.get("/reports", response_model=List[Dict[str, Any]])
def get_reports():
    store = _get_store()
    return list_reports(store)


@router.get("/report/{report_id}", response_model=LintReport)
def get_report(report_id: str):
    store = _get_store()
    try:
        return load_report(store, report_id)
    except FileNotFoundError:
        raise HTTPException(status_code=404, detail=f"Report {report_id} not found")


@router.post("/fix", response_model=Dict[str, Any])
def post_fix(body: FixBody):
    store = _get_store()
    try:
        report = load_report(store, body.report_id)
    except FileNotFoundError:
        raise HTTPException(status_code=404, detail=f"Report {body.report_id} not found")

    issue = next((i for i in report.issues if i.id == body.issue_id), None)
    if issue is None:
        raise HTTPException(status_code=404, detail=f"Issue {body.issue_id} not found")

    if body.action == "apply":
        try:
            apply_fix(store, issue, edit_payload=body.edit_payload)
        except FixError as exc:
            raise HTTPException(status_code=400, detail=str(exc))
        new_status = "fixed"
    elif body.action == "reject":
        new_status = "rejected"
    else:
        new_status = "wontfix"

    update_issue_status(store, body.report_id, body.issue_id, new_status)
    return {"id": body.issue_id, "status": new_status}
```

```tsx
import { useState } from "react";
import { AlertTriangle, AlertOctagon, Info, Check, X, Ban } from "lucide-react";
import type { LintIssue } from "../../types";

const SEVERITY_ICON = {
  error: AlertOctagon,
  warning: AlertTriangle,
  info: Info,
};

const SEVERITY_COLOR: Record<string, string> = {
  error: "text-red-600 dark:text-red-400",
  warning: "text-yellow-600 dark:text-yellow-400",
  info: "text-blue-500",
};

const STATUS_BADGE: Record<string, string> = {
  open: "bg-muted text-muted-foreground",
  fixed: "bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-300",
  rejected: "bg-red-100 text-red-800 dark:bg-red-900/30 dark:text-red-300",
  wontfix: "bg-zinc-200 text-zinc-700 dark:bg-zinc-800 dark:text-zinc-300",
};

interface LintIssueRowProps {
  issue: LintIssue;
  onApply: (issueId: string, editPayload?: { description: string }) => void;
  onReject: (issueId: string) => void;
  onWontFix: (issueId: string) => void;
  busy: boolean;
}

export function LintIssueRow({ issue, onApply, onReject, onWontFix, busy }: LintIssueRowProps) {
  const Icon = SEVERITY_ICON[issue.severity];
  const isEmptyDescriptionEdit =
    issue.type === "empty_description" && issue.auto_fixable;
  const [editText, setEditText] = useState("");
  const editValid = editText.trim().length >= 30;
  const closed = issue.status !== "open";

  return (
    <div className="border border-border rounded-md p-3 flex flex-col gap-2 text-xs bg-card">
      <div className="flex items-start gap-2">
        <Icon className={`w-4 h-4 mt-0.5 shrink-0 ${SEVERITY_COLOR[issue.severity]}`} />

            <span className="font-mono text-[11px] text-muted-foreground">·</span>
            <span className="font-mono text-[11px] text-muted-foreground truncate">
              {issue.location}
            </span>
            <span className={`ml-auto px-2 py-0.5 rounded text-[10px] uppercase ${STATUS_BADGE[issue.status]}`}>
              {issue.status}
            </span>
          </div>
          <p className="mt-1 text-foreground leading-snug">{issue.description}</p>
          {issue.suggested_fix && (
            <p className="mt-1 text-muted-foreground">
              suggested: {issue.suggested_fix}
            </p>
          )}
        </div>
      </div>

      {!closed && isEmptyDescriptionEdit && (
        <textarea
          value={editText}
          onChange={(e) => setEditText(e.target.value)}
          placeholder="Replacement description (≥30 chars)…"

            onClick={() => onReject(issue.id)}
            className="flex items-center gap-1 px-2 py-1 text-xs rounded bg-muted text-muted-foreground hover:text-foreground transition-colors disabled:opacity-40"
          >
            <X className="w-3 h-3" /> Reject
          </button>
          <button
            disabled={busy}
            onClick={() => onWontFix(issue.id)}
            className="flex items-center gap-1 px-2 py-1 text-xs rounded bg-muted text-muted-foreground hover:text-foreground transition-colors disabled:opacity-40"
          >
            <Ban className="w-3 h-3" /> Won't fix
          </button>
        </div>
      )}
    </div>
  );
}
```

```tsx
import { useEffect, useState } from "react";
import { Play, RefreshCw, ShieldCheck } from "lucide-react";
import { api } from "../../api/client";
import type { IssueSeverity, IssueStatus, LintReport } from "../../types";
import { LintIssueRow } from "./LintIssueRow";

type SeverityFilter = "all" | IssueSeverity;
type StatusFilter = "all" | IssueStatus;

export function LintPanel() {
  const [report, setReport] = useState<LintReport | null>(null);
  const [running, setRunning] = useState(false);
  const [busyIssueId, setBusyIssueId] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [severityFilter, setSeverityFilter] = useState<SeverityFilter>("all");
  const [statusFilter, setStatusFilter] = useState<StatusFilter>("open");

  async function loadLatestReport() {
    try {
      const summaries = await api.listLintReports();
      if (summaries.length === 0) {
        setReport(null);
        return;
      }
      const latest = await api.getLintReport(summaries[0].id);
      setReport(latest);
    } catch (e) {
      setError((e as Error).message);
    }
  }

  useEffect(() => {
    loadLatestReport();
  }, []);

  async function runNow() {
    setRunning(true);
    setError(null);
    try {
      const fresh = await api.runLint();
      setReport(fresh);
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setRunning(false);
    }
  }

  async function applyAction(
    issueId: string,
    action: "apply" | "reject" | "wontfix",
    editPayload?: { description: string },
  ) {
    if (!report) return;
    setBusyIssueId(issueId);
    setError(null);
    try {
      await api.applyLintFix({
        report_id: report.id,
        issue_id: issueId,
        action,
        edit_payload: editPayload,
      });
      const refreshed = await api.getLintReport(report.id);
      setReport(refreshed);
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setBusyIssueId(null);
    }
  }

  const issues = report?.issues ?? [];
  const filtered = issues.filter((i) => {
    if (severityFilter !== "all" && i.severity !== severityFilter) return false;
    if (statusFilter !== "all" && i.status !== statusFilter) return false;
    return true;
  });

  const sev = report?.counts_by_severity ?? { error: 0, warning: 0, info: 0 };

  return (
    <div className="h-full flex flex-col overflow-hidden">
      {/* Top bar */}
      <div className="flex items-center gap-3 px-4 py-3 border-b border-border shrink-0">
        <ShieldCheck className="w-4 h-4 text-primary" />
        <h2 className="text-sm font-medium">Lint</h2>
        <button
          onClick={runNow}
          disabled={running}
          className="flex items-center gap-1.5 ml-auto px-3 py-1.5 text-xs rounded-md bg-primary text-primary-foreground hover:opacity-90 transition-opacity disabled:opacity-40"
        >
          {running ? <RefreshCw className="w-3.5 h-3.5 animate-spin" /> : <Play className="w-3.5 h-3.5" />}
          {running ? "Running…" : "Run Now"}
        </button>
      </div>

      {/* Summary + filters */}
      <div className="flex items-center gap-3 px-4 py-2 border-b border-border shrink-0 flex-wrap">
        {report ? (
          <span className="text-xs text-muted-foreground">
            Report{" "}
            <span className="font-mono text-foreground">{report.id}</span> ·{" "}
            <span className="text-red-600">{sev.error} errors</span> ·{" "}
            <span className="text-yellow-600">{sev.warning} warnings</span> ·{" "}
            {issues.length} total
          </span>
        ) : (
          <span className="text-xs text-muted-foreground">
            No lint reports yet. Click <span className="font-medium">Run Now</span>.
          </span>
        )}

        <div className="flex gap-1 ml-auto">
          {(["all", "error", "warning"] as SeverityFilter[]).map((s) => (
            <button
              key={s}
              onClick={() => setSeverityFilter(s)}
              className={`px-2 py-0.5 rounded text-xs capitalize transition-colors ${
                severityFilter === s
                  ? "bg-primary/15 text-primary font-medium"
                  : "bg-muted text-muted-foreground"
              }`}
            >
              {s}
            </button>
          ))}
        </div>
        <span className="text-muted-foreground text-xs">|</span>
        <div className="flex gap-1">
          {(["open", "fixed", "rejected", "wontfix", "all"] as StatusFilter[]).map((s) => (
            <button
              key={s}
              onClick={() => setStatusFilter(s)}
              className={`px-2 py-0.5 rounded text-xs capitalize transition-colors ${
                statusFilter === s
                  ? "bg-primary/15 text-primary font-medium"
                  : "bg-muted text-muted-foreground"
              }`}
            >
              {s}
            </button>
          ))}
        </div>
      </div>

      {error && (
        <div className="px-4 py-2 text-xs text-red-600 bg-red-100/40 border-b border-red-200">
          {error}
        </div>
      )}

      {/* Issue list */}
      <div className="flex-1 overflow-auto p-4 flex flex-col gap-2">
        {report && filtered.length === 0 && (
          <div className="text-center text-muted-foreground text-sm py-12">
            {issues.length === 0 ? "No issues found." : "No issues match the current filter."}
          </div>
        )}
        {filtered.map((issue) => (
          <LintIssueRow
            key={issue.id}
            issue={issue}
            busy={busyIssueId === issue.id}
            onApply={(id, payload) => applyAction(id, "apply", payload)}
            onReject={(id) => applyAction(id, "reject")}
            onWontFix={(id) => applyAction(id, "wontfix")}
          />
        ))}
      </div>
    </div>
  );
}
```
